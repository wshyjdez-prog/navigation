    const double lat=30.593,lon=114.305;const auto before=QFileInfo(path).size();

    for(const auto& q:queries){
        const auto want=searchNearestPoisByScan(path,q,lat,lon,20);std::vector<double> ms;quint64 cand=0;bool same=true;int n=0;

    }
    std::vector<double> scan;for(int r=0;r<repeats;++r){QElapsedTimer t;t.start();searchNearestPoisByScan(path,"超市",lat,lon,20);scan.push_back(t.nsecsElapsed()/1e6);}
    std::sort(scan.begin(),scan.end());std::printf("%s,scan-baseline,超市,%.3f\n",qPrintable(name),scan[scan.size()/2]);

}
