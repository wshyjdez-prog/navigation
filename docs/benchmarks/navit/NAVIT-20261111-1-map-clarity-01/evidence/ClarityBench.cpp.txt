// Temporary benchmark: map clarity levels. Usage: CACHE TILE_ROOT low|medium|high RUNS SHOTDIR
#include "ui/map/TilePyramid.h"
#include "ui/map/MapFrame.h"
#include "contracts/ImportedDataset.h"
#include <QApplication>
#include <QDirIterator>
#include <QElapsedTimer>
#include <QEventLoop>
#include <QTimer>
#include <cmath>
#include <cstdio>
using namespace p1;
constexpr double world=40075016.68557849;
int main(int argc,char** argv){
    QApplication app(argc,argv);if(argc<6)return 2;
    const auto entry=data::ImportedDataset(QString::fromLocal8Bit(argv[1])).summary();const QString root=argv[2],name=argv[3],shots=argv[5];const int runs=std::atoi(argv[4]);
    const auto clarity=name=="high"?ui::MapClarity::High:name=="medium"?ui::MapClarity::Medium:ui::MapClarity::Low;const QSize size(1280,800);
    for(int z:{13,15,17})for(int run=0;run<runs;++run){
        {QDirIterator it(root,{"raster-style*"},QDir::Dirs,QDirIterator::Subdirectories);QStringList d;while(it.hasNext())d<<it.next();for(auto&x:d)QDir(x).removeRecursively();}
        ui::TilePyramid pyramid(nullptr,root);pyramid.setDataset(entry,{});pyramid.setClarity(clarity);
        const auto c=ui::mapProject({114.305,30.593});const double w=world*size.width()/(256*std::ldexp(1.,z))*.75;const QRectF view(c.x()-w/2,c.y()-w*size.height()/size.width()/2,w,w*size.height()/size.width());
        auto once=[&](QImage* shot)->std::pair<double,ui::PyramidStats>{QEventLoop loop;QElapsedTimer t;double ms=-1;ui::PyramidStats out;
            auto con=QObject::connect(&pyramid,&ui::TilePyramid::frameReady,&loop,[&](QImage im,QRectF fv,ui::PyramidStats st){if(st.complete){ms=t.nsecsElapsed()/1e6;out=st;if(shot)*shot=ui::reprojectMapFrame(im,fv,view,size,Qt::magenta);loop.quit();}});
            QTimer::singleShot(600000,&loop,&QEventLoop::quit);t.start();pyramid.request(view,size,false);loop.exec();QObject::disconnect(con);return {ms,out};};
        QImage shot;auto [cold,st]=once(run==0?&shot:nullptr);
        qint64 bytes=0;int files=0;{QDirIterator it(root,{"*.png"},QDir::Files,QDirIterator::Subdirectories);while(it.hasNext()){it.next();bytes+=it.fileInfo().size();++files;}}
        auto [warm,st2]=once(nullptr);
        std::printf("%s,z%d,%d,%d,%.1f,%.1f,%d,%lld\n",qPrintable(name),z,st.zoom,run,cold,warm,files,(long long)bytes);std::fflush(stdout);
        if(run==0&&!shots.isEmpty()){QDir().mkpath(shots);shot.save(QString("%1/%2-z%3.png").arg(shots,name).arg(z));}
    }
    return 0;
}
