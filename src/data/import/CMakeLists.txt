find_package(SQLite3 REQUIRED GLOBAL)
find_package(EXPAT REQUIRED GLOBAL)
find_package(ZLIB REQUIRED GLOBAL)
find_package(BZip2 REQUIRED GLOBAL)
find_package(Threads REQUIRED GLOBAL)
find_path(OSMIUM_INCLUDE_DIR osmium/version.hpp REQUIRED)
find_path(PROTOZERO_INCLUDE_DIR protozero/version.hpp REQUIRED)
file(STRINGS "${OSMIUM_INCLUDE_DIR}/osmium/version.hpp" OSMIUM_VERSION_LINE
    REGEX "^#define LIBOSMIUM_VERSION_STRING")
if(NOT OSMIUM_VERSION_LINE MATCHES "\"2\\.20\\.0\"")
    message(FATAL_ERROR "This import baseline requires libosmium 2.20.0; found: ${OSMIUM_VERSION_LINE}")
endif()
add_library(p1_data_import STATIC OsmImportParser.cpp ImportedDataset.cpp ImportedRoads.cpp)
target_include_directories(p1_data_import PUBLIC "${PROJECT_SOURCE_DIR}/src" "${PROJECT_SOURCE_DIR}/include")
target_include_directories(p1_data_import SYSTEM PRIVATE "${OSMIUM_INCLUDE_DIR}" "${PROTOZERO_INCLUDE_DIR}")
target_link_libraries(p1_data_import PUBLIC Qt6::Core p1_data_osmx PRIVATE SQLite::SQLite3 EXPAT::EXPAT ZLIB::ZLIB BZip2::BZip2 Threads::Threads)
target_compile_options(p1_data_import PRIVATE "$<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic>" "$<$<CXX_COMPILER_ID:MSVC>:/utf-8>")
add_executable(p1-import-cli tools/ImportCli.cpp)
target_link_libraries(p1-import-cli PRIVATE p1_data_import)
if(BUILD_TESTING)
    qt_add_executable(p1-import-tests tests/ImportTests.cpp)
    target_include_directories(p1-import-tests SYSTEM PRIVATE "${OSMIUM_INCLUDE_DIR}" "${PROTOZERO_INCLUDE_DIR}")
    target_link_libraries(p1-import-tests PRIVATE p1_data_import Qt6::Test EXPAT::EXPAT ZLIB::ZLIB BZip2::BZip2 Threads::Threads)
    add_test(NAME data-import COMMAND p1-import-tests)
endif()
